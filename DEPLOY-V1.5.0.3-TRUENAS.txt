1. Extract this complete release and upload its contents to the main branch of:
   https://github.com/valianktni1/march-weddings-by-mark-accounts

2. In the TrueNAS shell, run these commands exactly:

cd /mnt/apps/dockge/data/accounts-2026-use

ACCOUNTS_BACKUP="/mnt/apps/dockge/data/accounts-2026-use/accounts-before-v1503-$(date +%Y%m%d-%H%M%S).archive"

docker exec accounts-2026-db sh -lc \
  'mongodump --username "$MONGO_INITDB_ROOT_USERNAME" --password "$MONGO_INITDB_ROOT_PASSWORD" --authenticationDatabase admin --db weddings_by_mark_accounts_2026 --archive' \
  > "$ACCOUNTS_BACKUP"

ACCOUNTS_COMPOSE=/mnt/apps/dockge/data/accounts-2026-use/compose.yaml

docker compose -f "$ACCOUNTS_COMPOSE" build --no-cache backend frontend

docker compose -f "$ACCOUNTS_COMPOSE" ps
curl -fsS http://127.0.0.1:30053/api/health
